# EdSight VPS Deployment - Copy and Paste These Commands

# 1. SSH into VPS
ssh root@72.61.140.217

# 2. Clone repository
cd /var/www
git clone https://github.com/isaackcz/Edsight.git edsight
cd edsight

# 3. Install Docker (if not installed)
apt update && apt upgrade -y
curl -fsSL https://get.docker.com -o get-docker.sh
sh get-docker.sh
curl -L "https://github.com/docker/compose/releases/latest/download/docker-compose-$(uname -s)-$(uname -m)" -o /usr/local/bin/docker-compose
chmod +x /usr/local/bin/docker-compose

# 4. Generate secure passwords
SECRET_KEY=$(python3 -c "import secrets; print(secrets.token_urlsafe(50))" 2>/dev/null || openssl rand -base64 50)
DB_PASSWORD=$(openssl rand -base64 32 | head -c 32)
MYSQL_ROOT_PASSWORD=$(openssl rand -base64 32 | head -c 32)

# 5. Create .env.production file
cat > .env.production << EOF
SECRET_KEY=$SECRET_KEY
DEBUG=False
ALLOWED_HOSTS=72.61.140.217,*
DB_HOST=mysql
DB_PORT=3306
DB_NAME=edsight_prod
DB_USER=edsight_prod
DB_PASSWORD=$DB_PASSWORD
REDIS_HOST=redis
REDIS_PORT=6379
REDIS_DB=0
CELERY_BROKER_URL=redis://redis:6379/0
CELERY_RESULT_BACKEND=redis://redis:6379/0
CELERY_TASK_ALWAYS_EAGER=False
STATIC_ROOT=/vol/static
DJANGO_HOST=django
REDIS_URL=redis://redis:6379/0
MYSQL_ROOT_PASSWORD=$MYSQL_ROOT_PASSWORD
MYSQL_DATABASE=edsight_prod
MYSQL_USER=edsight_prod
MYSQL_PASSWORD=$DB_PASSWORD
GUNICORN_WORKERS=4
SESSION_COOKIE_SECURE=True
CSRF_COOKIE_SECURE=True
SECURE_SSL_REDIRECT=True
SECURE_HSTS_SECONDS=31536000
SECURE_HSTS_INCLUDE_SUBDOMAINS=True
SECURE_HSTS_PRELOAD=True
EOF

# 6. Build Docker images (25-35 minutes)
docker-compose -f docker-compose.yml -f docker-compose.prod.yml build

# 7. Start services
docker-compose -f docker-compose.yml -f docker-compose.prod.yml up -d

# 8. Check status
docker-compose ps

# 9. View logs
docker-compose logs -f
